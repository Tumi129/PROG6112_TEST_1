2.1   public interface IConsoles { 

    String getConsoleType(); 

    String getStore(); 

    int getTotalSales(); 

} 

2.2 public abstract class ConsoleSales implements IConsoles { 

    private String consoleType; 

    private String store; 

    private int total; 

  

    public ConsoleSales(String consoleType, String store, int total) { 

        this.consoleType = consoleType; 

        this.store = store; 

        this.total = total; 

    } 

  

    @Override 

    public String getConsoleType() { return consoleType; } 

  

    @Override 

    public String getStore() { return store; } 

  

    @Override 

    public int getTotalSales() { return total; } 

} 

2.3 public class Number1ElectronicReport extends ConsoleSales { 

public Number1ElectronicReport(String consoleType, String store, int total) { 
    super(consoleType, store, total); 
} 
 
public void printNumber1ElectronicStoreReport() { 
    System.out.println("NUMBER 1 ELECTRONIC STORE REPORT"); 
    System.out.println("************************"); 
    System.out.println("CONSOLE TYPE: " + getConsoleType()); 
    System.out.println("STORE: " + getStore()); 
    System.out.println("SALES TOTAL: " + getTotalSales()); 
    System.out.println("************************"); 
} 
  

} 

2.4  import java.util.Scanner; 

  

public class RunApplication { 

    public static void main(String[] args) { 

        Scanner input = new Scanner(System.in); 

  

        System.out.print("Enter the console type: "); 

        String type = input.nextLine(); 

  

        System.out.print("Enter the store for the console: "); 

        String store = input.nextLine(); 

  

        System.out.print("Enter the total " + type + " sales for " + store + ": "); 

        int total = input.nextInt(); 

  

        Number1ElectronicReport report = new Number1ElectronicReport(type, store, total); 

        report.printNumber1ElectronicStoreReport(); 

    } 

}
